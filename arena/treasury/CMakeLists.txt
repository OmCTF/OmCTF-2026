cmake_minimum_required(VERSION 3.20)
project(kazna LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(kazna-referee backend/referee.cpp backend/arena.cpp backend/strategy.cpp)
target_include_directories(kazna-referee PRIVATE backend backend/vendor sdk/include)
find_package(OpenSSL REQUIRED)
target_link_libraries(kazna-referee PRIVATE pthread OpenSSL::Crypto)
target_compile_options(kazna-referee PRIVATE -O2 -Wall -Wextra -Wpedantic -fstack-protector-strong -fPIE)
target_link_options(kazna-referee PRIVATE -pie -Wl,-z,relro,-z,now)
enable_testing()
add_executable(kazna-arena-test tests/arena_test.cpp backend/arena.cpp)
target_include_directories(kazna-arena-test PRIVATE backend backend/vendor sdk/include)
add_test(NAME arena-rules COMMAND kazna-arena-test)
